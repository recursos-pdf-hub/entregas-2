    var link = e.target.closest('a[href^="#"]');
    if (!link) return;
    var id = link.getAttribute("href");
    if (id.length < 2) return;
    var target = document.querySelector(id);
    if (!target) return;
    e.preventDefault();
    var top = target.getBoundingClientRect().top + window.pageYOffset - 76;
    window.scrollTo({ top: top, behavior: reduceMotion ? "auto" : "smooth" });
    target.setAttribute("tabindex", "-1");
    target.focus({ preventScroll: true });
  });

  /* ---------- ACCORDIONS (sumário + FAQ) ---------- */
  document.querySelectorAll("[data-accordion]").forEach(function (group) {
    group.querySelectorAll(".ps-acc__trigger").forEach(function (trigger) {
      var panel = document.getElementById(trigger.getAttribute("aria-controls"));
      trigger.addEventListener("click", function () {
        var isOpen = trigger.getAttribute("aria-expanded") === "true";
        if (isOpen) {
          panel.style.height = panel.scrollHeight + "px";
          requestAnimationFrame(function () { panel.style.height = "0px"; });
          trigger.setAttribute("aria-expanded", "false");
        } else {
          trigger.setAttribute("aria-expanded", "true");
          panel.style.height = panel.scrollHeight + "px";
          window.setTimeout(function () {
            if (trigger.getAttribute("aria-expanded") === "true") panel.style.height = "auto";
          }, 360);
        }
      });
    });
  });

  /* ---------- GALERIA EM LOOP INFINITO ---------- */
  var marquee = document.getElementById("ps-marquee");
  var track = document.getElementById("ps-gallery");
  if (marquee && track && track.firstElementChild) {
    // 1. agrupa os itens originais
    var group = document.createElement("div");
    group.className = "ps-gallery__group";
    while (track.firstElementChild) group.appendChild(track.firstElementChild);
    track.appendChild(group);

    // 2. imagens deixam de ser lazy: a largura não pode mudar depois
    track.querySelectorAll("img").forEach(function (img) {
      img.loading = "eager";
      img.removeAttribute("loading");
    });

    var groupWidth = 0;
    var x = 0;
    var speed = 42;   // pixels por segundo
    var last = null;
    var paused = false;

    // 3. clona grupos até o trilho cobrir pelo menos 3x a largura da tela.
    //    com folga assim, nunca existe borda visível durante o reposicionamento.
    function build() {
      groupWidth = group.getBoundingClientRect().width;
      if (!groupWidth) return;
      var alvo = window.innerWidth * 3 + groupWidth;
      var atual = track.getBoundingClientRect().width;
      var guarda = 0;
      while (atual < alvo && guarda < 12) {
        var c = group.cloneNode(true);
        c.setAttribute("aria-hidden", "true");
        track.appendChild(c);
        atual += groupWidth;
        guarda++;
      }
    }

    function measure() {
      var novo = group.getBoundingClientRect().width;
      if (novo && Math.abs(novo - groupWidth) > 1) {
        groupWidth = novo;
        x = 0;
      }
    }

    build();
    window.addEventListener("resize", function () { measure(); build(); });
    window.addEventListener("orientationchange", function () { measure(); build(); });
    window.addEventListener("load", function () { measure(); build(); });
    track.querySelectorAll("img").forEach(function (img) {
      if (!img.complete) img.addEventListener("load", function () { measure(); build(); });
    });

    // 4. loop em pixels
    function frame(now) {
      if (last === null) last = now;
      var dt = (now - last) / 1000;
      last = now;
      if (dt > 0.25) dt = 0;
      if (!paused && groupWidth > 0) {
        x -= speed * dt;
        while (x <= -groupWidth) x += groupWidth;
        track.style.transform = "translate3d(" + x + "px,0,0)";
      }
      requestAnimationFrame(frame);
    }
    if (!reduceMotion) requestAnimationFrame(frame);

    marquee.addEventListener("click", function () { paused = !paused; });
    marquee.addEventListener("touchstart", function () { paused = true; }, { passive: true });
    marquee.addEventListener("touchend", function () { paused = false; }, { passive: true });
    marquee.addEventListener("mouseenter", function () { paused = true; });
    marquee.addEventListener("mouseleave", function () { paused = false; });
    document.addEventListener("visibilitychange", function () { last = null; });
  }

  /* ---------- REVEAL AO ENTRAR NO VIEWPORT ---------- */
  var revealables = document.querySelectorAll(".ps-reveal");
  if (reduceMotion || !("IntersectionObserver" in window)) {
    revealables.forEach(function (el) { el.classList.add("ps-is-visible"); });
  } else {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add("ps-is-visible");
          io.unobserve(entry.target);
        }
      });
    }, { rootMargin: "0px 0px -8% 0px", threshold: 0.08 });
    revealables.forEach(function (el) { io.observe(el); });
  }
})();
